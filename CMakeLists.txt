cmake_minimum_required(VERSION 3.24)
project(AviSynthComposite VERSION 0.1.0 LANGUAGES C CXX)

option(CP_BUILD_TESTS "Build independent scalar, SIMD and C API tests" ${PROJECT_IS_TOP_LEVEL})
option(CP_SANITIZERS "Enable address and undefined behavior sanitizers (Clang/GCC)" OFF)
option(CP_SCALAR_ONLY "Build without Highway SIMD targets or runtime dependency" OFF)
option(CP_BUILD_BENCHMARKS "Build deterministic kernel microbenchmarks" OFF)

if(NOT CP_SCALAR_ONLY AND NOT TARGET hwy)
  set(HWY_ENABLE_CONTRIB OFF CACHE BOOL "" FORCE)
  set(HWY_ENABLE_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(HWY_ENABLE_TESTS OFF CACHE BOOL "" FORCE)
  set(HWY_ENABLE_TOOLS OFF CACHE BOOL "" FORCE)
  set(HWY_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
  set(HWY_LIBRARY_TYPE STATIC CACHE STRING "" FORCE)
  set(HWY_FORCE_STATIC_LIBS ON CACHE BOOL "" FORCE)
  set(HWY_CMAKE_RVV OFF CACHE BOOL "" FORCE)
  add_subdirectory(third_party/highway EXCLUDE_FROM_ALL)
endif()

# Parent projects may enable contraction globally. Strict reference builds must
# not inherit MSVC's mutually exclusive /fp:contract option.
if(MSVC AND CP_BUILD_TESTS)
  get_directory_property(cp_inherited_options COMPILE_OPTIONS)
  list(FILTER cp_inherited_options EXCLUDE REGEX "^/fp:contract$")
  set_property(DIRECTORY PROPERTY COMPILE_OPTIONS "${cp_inherited_options}")
endif()

add_library(Composite STATIC src/plane.cpp src/yuv.cpp src/mask.cpp src/utilities.cpp src/highway.cpp src/highway_extra-inl.h)
add_library(AviSynth::Composite ALIAS Composite)
target_compile_features(Composite PRIVATE cxx_std_17)
set_target_properties(Composite PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(Composite PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>)
target_include_directories(Composite PRIVATE src)
if(CP_SCALAR_ONLY)
  target_compile_definitions(Composite PRIVATE CP_SCALAR_ONLY)
else()
  target_link_libraries(Composite PRIVATE hwy)
  # Keep diagnostics for project code; external Highway headers use extensions.
  target_include_directories(Composite SYSTEM PRIVATE $<TARGET_PROPERTY:hwy,INTERFACE_INCLUDE_DIRECTORIES>)
endif()
if(MSVC)
  target_compile_options(Composite PRIVATE /W4)
  if(CP_BUILD_TESTS)
    target_compile_options(Composite PRIVATE /fp:strict)
  else()
    target_compile_options(Composite PRIVATE /fp:precise /fp:contract)
  endif()
else()
  target_compile_options(Composite PRIVATE -Wall -Wextra -Wpedantic)
  if(CP_BUILD_TESTS)
    target_compile_options(Composite PRIVATE -ffp-contract=off)
  else()
    target_compile_options(Composite PRIVATE -ffp-contract=fast)
  endif()
endif()
if(CP_SANITIZERS)
  if(MSVC OR NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    message(FATAL_ERROR "CP_SANITIZERS requires a non-MSVC Clang/GCC toolchain")
  endif()
  target_compile_options(Composite PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(Composite PUBLIC -fsanitize=address,undefined)
endif()
if(CP_BUILD_TESTS)
  enable_testing()
  add_subdirectory(tests)
endif()
if(CP_BUILD_BENCHMARKS)
  add_subdirectory(benchmarks)
endif()
