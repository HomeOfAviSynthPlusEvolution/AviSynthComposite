set(CP_UPSTREAM_BENCH_DIR "" CACHE PATH "Temporary headers generated by prepare_upstream.py")
add_executable(composite_bench bench.cpp)
add_executable(composite_average_bench average_bench.cpp)
target_compile_features(composite_average_bench PRIVATE cxx_std_17)
target_link_libraries(composite_average_bench PRIVATE AviSynth::Composite)
target_compile_features(composite_bench PRIVATE cxx_std_17)
target_link_libraries(composite_bench PRIVATE AviSynth::Composite)
if(MSVC)
  target_compile_options(composite_bench PRIVATE /W4 /fp:strict)
  target_compile_options(composite_average_bench PRIVATE /W4 /fp:strict)
else()
  target_compile_options(composite_bench PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
  target_compile_options(composite_average_bench PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
endif()
if(CP_UPSTREAM_BENCH_DIR)
  if(NOT EXISTS "${CP_UPSTREAM_BENCH_DIR}/layer_kernels.h")
    message(FATAL_ERROR "Run benchmarks/prepare_upstream.py first")
  endif()
  target_include_directories(composite_bench SYSTEM PRIVATE "${CP_UPSTREAM_BENCH_DIR}")
  target_compile_definitions(composite_bench PRIVATE CP_BENCH_UPSTREAM)
  if(NOT EXISTS "${CP_UPSTREAM_BENCH_DIR}/yuv_add.cpp")
    message(FATAL_ERROR "Re-run prepare_upstream.py for integer YUV Add/Subtract")
  endif()
  target_sources(composite_bench PRIVATE "${CP_UPSTREAM_BENCH_DIR}/yuv_add.cpp")
  if(CMAKE_SYSTEM_PROCESSOR MATCHES "AMD64|amd64|x86_64|i.86|x86")
    if(NOT EXISTS "${CP_UPSTREAM_BENCH_DIR}/average_avx2.cpp")
      message(FATAL_ERROR "Re-run prepare_upstream.py to extract the AVX2 average reference")
    endif()
    target_sources(composite_bench PRIVATE "${CP_UPSTREAM_BENCH_DIR}/average_avx2.cpp")
    if(NOT EXISTS "${CP_UPSTREAM_BENCH_DIR}/weighted_avx2.cpp")
      message(FATAL_ERROR "Re-run prepare_upstream.py to extract the weighted AVX2 reference")
    endif()
    target_sources(composite_bench PRIVATE "${CP_UPSTREAM_BENCH_DIR}/weighted_avx2.cpp")
    if(NOT EXISTS "${CP_UPSTREAM_BENCH_DIR}/masked_avx2.cpp")
      message(FATAL_ERROR "Re-run prepare_upstream.py for masked AVX2")
    endif()
    target_sources(composite_bench PRIVATE "${CP_UPSTREAM_BENCH_DIR}/masked_avx2.cpp")
    if(NOT EXISTS "${CP_UPSTREAM_BENCH_DIR}/overlay_mul_avx2.cpp")
      message(FATAL_ERROR "Re-run prepare_upstream.py for Overlay Multiply")
    endif()
    target_sources(composite_bench PRIVATE "${CP_UPSTREAM_BENCH_DIR}/overlay_mul_avx2.cpp")
    target_sources(composite_average_bench PRIVATE "${CP_UPSTREAM_BENCH_DIR}/average_avx2.cpp")
    if(MSVC)
      set_source_files_properties("${CP_UPSTREAM_BENCH_DIR}/average_avx2.cpp" PROPERTIES COMPILE_OPTIONS /arch:AVX2)
      set_source_files_properties("${CP_UPSTREAM_BENCH_DIR}/weighted_avx2.cpp" PROPERTIES COMPILE_OPTIONS /arch:AVX2)
      set_source_files_properties("${CP_UPSTREAM_BENCH_DIR}/masked_avx2.cpp" "${CP_UPSTREAM_BENCH_DIR}/overlay_mul_avx2.cpp" PROPERTIES COMPILE_OPTIONS /arch:AVX2)
    else()
      set_source_files_properties("${CP_UPSTREAM_BENCH_DIR}/average_avx2.cpp" PROPERTIES COMPILE_OPTIONS -mavx2)
      set_source_files_properties("${CP_UPSTREAM_BENCH_DIR}/weighted_avx2.cpp" PROPERTIES COMPILE_OPTIONS "-mavx2;-mfma")
      set_source_files_properties("${CP_UPSTREAM_BENCH_DIR}/masked_avx2.cpp" "${CP_UPSTREAM_BENCH_DIR}/overlay_mul_avx2.cpp" PROPERTIES COMPILE_OPTIONS "-mavx2;-mfma")
    endif()
    target_compile_definitions(composite_bench PRIVATE CP_BENCH_AVX2)
    target_compile_definitions(composite_average_bench PRIVATE CP_BENCH_AVX2)
    target_include_directories(composite_average_bench SYSTEM PRIVATE "${CP_UPSTREAM_BENCH_DIR}")
  endif()
endif()
